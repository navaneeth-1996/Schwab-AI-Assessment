# Run with: python -m course_registration.cli --capacity 1 < examples/demo.txt
# One seat; prerequisite CS101. dave has not completed it.
learner amy CS101
learner ben CS101
learner cara CS101
learner dan CS101
learner dave

enroll amy
enroll dave
enroll ben
enroll cara
enroll dan
enroll ben
show

# ben's CS101 record is revoked while he waits (safeguard case)
revoke ben CS101
drop amy
show

# ben's record is corrected; he is still first in line and gets the next seat
complete ben CS101
drop cara
show

# amy dropped earlier: a seat can be held only once, so she cannot enroll again
enroll amy
# dan leaves the waitlist; he never held a seat, so he may re-join (at the back)
leave dan
enroll dan
drop amy
drop nobody
show
